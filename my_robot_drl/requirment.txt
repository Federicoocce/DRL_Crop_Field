# Update system and install prerequisite tools
sudo apt update -y && sudo apt upgrade -y
sudo apt install -y git python3-pip curl software-properties-common locales -y

# Configure system locale for ROS 2
sudo locale-gen en_US en_US.UTF-8 -y
sudo update-locale LC_ALL=en_US.UTF-8 LANG=en_US.UTF-8 -y

# Add the ROS 2 repository
sudo add-apt-repository universe -y
sudo curl -sSL https://raw.githubusercontent.com/ros/rosdistro/master/ros.key -o /usr/share/keyrings/ros-archive-keyring.gpg 
echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/ros-archive-keyring.gpg] http://packages.ros.org/ros2/ubuntu $(. /etc/os-release && echo $UBUNTU_CODENAME) main" | sudo tee /etc/apt/sources.list.d/ros2.list > /dev/null

# Install ROS 2 Humble Desktop and developer tools
sudo apt update -y
sudo apt install -y ros-humble-desktop ros-dev-tools

# Install all required Python packages for the system's Python interpreter
# This includes the fixes for NumPy, PyTorch (CUDA 12.x), and lxml
sudo pip3 install \
    "stable-baselines3[extra]" \
    gymnasium \
    PyYAML \
    Pillow \
    opencv-python \
    matplotlib \
    lxml \
    "numpy<2" \
    git+https://github.com/rdesc/pydubins.git \
    wandb

# Install PyTorch with CUDA 12.1 support
sudo pip3 install torch torchvision torchaudio 

# Create the workspace and clone the repository
mkdir -p ~/ros2_ws/src
cd ~/ros2_ws/src
git clone -b transformer-implementation-rear-camera https://github.com/Federicoocce/DRL_Crop_Field.git

# Install ROS dependencies and build the workspace
cd ~/ros2_ws
sudo rosdep init || echo "rosdep already initialized."
rosdep update
rosdep install -i --from-path src --rosdistro humble -y
echo "" >> /root/.bashrc
echo "# DRL_Crop_Field Project Configuration" >> /root/.bashrc
echo "export LIBGL_ALWAYS_SOFTWARE=1" >> /root/.bashrc
echo "source /opt/ros/humble/setup.bash" >> /root/.bashrc
echo "source /usr/share/gazebo/setup.bash" >> /root/.bashrc
echo "source /root/ros2_ws/install/setup.bash" >> /root/.bashrc

echo "--- VAST.AI SETUP COMPLETE ---"
source /opt/ros/humble/setup.bash
source /usr/share/gazebo/setup.bash
source ~/ros2_ws/install/setup.bash
colcon build
source ~/ros2_ws/install/setup.bash
echo "source ~/ros2_ws/install/setup.bash" >> ~/.bashrc
export LIBGL_ALWAYS_SOFTWARE=1
source /opt/ros/humble/setup.bash
source /usr/share/gazebo/setup.bash
source ~/ros2_ws/install/setup.bash
pip install --upgrade protobuf
wandb login 
ros2 launch my_robot_drl train_maize_nav.launch.py headless:=False mode:=il
