#!/bin/bash
set -e

# --- 1. System Setup & ROS 2 ---
echo "--> Updating system and installing dependencies..."
sudo apt-get update -y
sudo apt-get install -y git python3-pip curl software-properties-common locales apt-utils libgl1 -y

# Install ROS 2 Humble if not found
if ! command -v ros2 &> /dev/null; then
    sudo add-apt-repository universe -y
    sudo curl -sSL https://raw.githubusercontent.com/ros/rosdistro/master/ros.key -o /usr/share/keyrings/ros-archive-keyring.gpg
    echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/ros-archive-keyring.gpg] http://packages.ros.org/ros2/ubuntu $(. /etc/os-release && echo $UBUNTU_CODENAME) main" | sudo tee /etc/apt/sources.list.d/ros2.list > /dev/null
    sudo apt-get update -y
    sudo apt-get install -y ros-humble-desktop ros-dev-tools
fi

# Set Locale
sudo locale-gen en_US en_US.UTF-8
sudo update-locale LC_ALL=en_US.UTF-8 LANG=en_US.UTF-8
export LANG=en_US.UTF-8

# --- 2. Python Environment & Dependencies ---
echo "--> configuring Python..."
sudo pip3 install --upgrade pip "setuptools>=60.0" wheel
sudo apt-get remove -y python3-sympy || true

echo "--> Installing ALL Dependencies (including missing MMDet utils)..."
# I added 'terminaltables', 'termcolor', 'pycocotools', 'addict', 'yapf' here.
# These are the ones that were skipped by --no-deps previously.
sudo pip3 install --ignore-installed \
    "numpy<2.0.0" \
    "scipy==1.12.0" \
    "scikit-image==0.22.0" \
    "matplotlib==3.5.1" \
    "protobuf>=3.20" \
    "yapf==0.40.1" \
    "addict" \
    "terminaltables" \
    "termcolor" \
    "pycocotools" \
    gymnasium \
    PyYAML \
    Pillow \
    "opencv-python~=4.8.0" \
    lxml \
    git+https://github.com/rdesc/pydubins.git \
    wandb \
    ujson \
    timm \
    "transformers==4.30.2" \
    open3d==0.18.0

# --- 3. Machine Learning Stack (Strict Versioning) ---
echo "--> Force-Installing PyTorch 1.12.0 (CUDA 11.3)..."
# We KEEP --no-deps here because otherwise pip will upgrade NumPy to 2.0
sudo pip3 install torch==1.12.0+cu113 torchvision==0.13.0+cu113 torchaudio==0.12.0 \
    --extra-index-url https://download.pytorch.org/whl/cu113 \
    --force-reinstall --no-deps

echo "--> Installing MMCV-Full (Matched Binary)..."
export CUDA_HOME=/usr/local/cuda
# We KEEP --no-deps here to protect PyTorch version, but we installed the deps manually above.
sudo pip3 install mmcv-full==1.7.1 -f https://download.openmmlab.com/mmcv/dist/cu113/torch1.12.0/index.html 
sudo pip3 install mmdet==2.28.2 

# Safeguard: Ensure numpy didn't drift
sudo pip3 install "numpy<2.0.0" --force-reinstall

# --- 4. System Linker Hacks (The "Nuclear" Fix) ---
echo "--> Applying System Linker Fixes for CUDA..."
TORCH_LIB_PATH=$(python3 -c "import torch; import os; print(os.path.join(os.path.dirname(torch.__file__), 'lib'))")

echo "Linking libs from: $TORCH_LIB_PATH"

# Create hard symlinks to /usr/lib so ROS can't miss them
sudo ln -sf "$TORCH_LIB_PATH/libcudart.so.11.0" /usr/lib/x86_64-linux-gnu/libcudart.so.11.0
sudo ln -sf "$TORCH_LIB_PATH/libcublas.so.11" /usr/lib/x86_64-linux-gnu/libcublas.so.11
sudo ln -sf "$TORCH_LIB_PATH/libcublasLt.so.11" /usr/lib/x86_64-linux-gnu/libcublasLt.so.11
sudo ln -sf "$TORCH_LIB_PATH/libcusolver.so.11" /usr/lib/x86_64-linux-gnu/libcusolver.so.11
sudo ln -sf "$TORCH_LIB_PATH/libcusparse.so.11" /usr/lib/x86_64-linux-gnu/libcusparse.so.11
sudo ln -sf "$TORCH_LIB_PATH/libnvrtc.so.11.2" /usr/lib/x86_64-linux-gnu/libnvrtc.so.11.2

# Update ldconfig
echo "$TORCH_LIB_PATH" | sudo tee /etc/ld.so.conf.d/torch_libs.conf
sudo ldconfig

# --- 5. Workspace Build ---
echo "--> Building ROS Workspace..."
mkdir -p ~/ros2_ws/src
cd ~/ros2_ws/src
rm -rf DRL_Crop_Field
git clone -b transfuser_auxiliary_tasks https://github.com/Federicoocce/DRL_Crop_Field.git

cd ~/ros2_ws
sudo rosdep init || echo "rosdep initialized"
rosdep update
rosdep install -i --from-path src --rosdistro humble -y

source /opt/ros/humble/setup.bash
colcon build

# --- 6. Launch ---
echo "--> Launching..."
source /opt/ros/humble/setup.bash
source /usr/share/gazebo/setup.bash
source ~/ros2_ws/install/setup.bash

export LD_LIBRARY_PATH=$LD_LIBRARY_PATH:$TORCH_LIB_PATH
export LIBGL_ALWAYS_SOFTWARE=1
export MPLBACKEND=Agg

# W&B Login
wandb login 

# Run
ros2 launch my_robot_drl train_maize_nav.launch.py headless:=False mode:=il
